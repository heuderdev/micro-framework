Receitas

Formulários completos para copiar e adaptar: cadastro PF/PJ com endereço, busca de CEP, login, troca de senha, pedido com itens, formulário em etapas, avatar, importação de planilha e formulário em modal.

Todas as receitas assumem os scripts carregados como em Instalação (núcleo + máscaras, e os módulos citados). Os endpoints são exemplos: troque pelos seus.

Preencher endereço pelo CEP

Acrescente ao cadastro acima: quando o CEP for válido, consulta o ViaCEP e preenche os campos (sem bloquear se o serviço falhar).

<form x-data="enderecoForm()" x-form action="/api/clientes" method="post"
      @form:validated="$event.detail.field === 'cep' && $event.detail.valid && buscarCep()" novalidate>
    <input name="cep" data-form-mask="cep" data-form-v="required|cep">
    <input name="rua" data-form-v="required">
    <input name="bairro">
    <input name="cidade" data-form-v="required">
    <input name="uf" data-form-v="required|length:2">
    <small x-show="buscando">Buscando endereço…</small>
</form>

<script>
function enderecoForm() {
    return {
        buscando: false,
        ultimo: null,
        async buscarCep() {
            const cep = this.$form.getValue('cep', { as: 'raw' })
            if (cep === this.ultimo) return
            this.ultimo = cep
            this.buscando = true
            try {
                const r = await fetch(`https://viacep.com.br/ws/${cep}/json/`)
                const d = await r.json()
                if (d.erro) return this.$form.setErrors({ cep: 'CEP não encontrado.' }, { replace: false })
                this.$form.setValue('rua', d.logradouro)
                this.$form.setValue('bairro', d.bairro)
                this.$form.setValue('cidade', d.localidade)
                this.$form.setValue('uf', d.uf)
            } catch (e) {
                // serviço fora: a pessoa preenche manualmente
            } finally {
                this.buscando = false
            }
        },
    }
}
</script>

Pedido com itens e total

<form x-data="pedido()" x-form action="/api/pedidos" method="post" data-form-encoding="json" data-form-idempotency novalidate>
    <template x-for="(item, i) in itens" :key="item.id">
        <div class="linha">
            <input :name="`itens[${i}][produto]`" x-model="item.produto" data-form-label="Produto" data-form-v="required|min:2">
            <input :name="`itens[${i}][quantidade]`" x-model.number="item.qtd" type="number" min="1"
                   data-form-label="Quantidade" data-form-v="required|integer|min:1|max:999">
            <input :name="`itens[${i}][preco]`" data-form-mask="currency" data-form-label="Preço"
                   data-form-v="required|min_value:0.01"
                   @input="item.preco = Number($form.getValue(`itens[${i}][preco]`) || 0)">
            <span x-text="(item.qtd * item.preco).toLocaleString('pt-BR', { style: 'currency', currency: 'BRL' })"></span>
            <button type="button" @click="itens.splice(i, 1)" :disabled="itens.length === 1">Remover</button>
        </div>
    </template>
    <button type="button" @click="itens.push({ id: ++seq, produto: '', qtd: 1, preco: 0 })">Adicionar item</button>

    <p>Total: <strong x-text="total().toLocaleString('pt-BR', { style: 'currency', currency: 'BRL' })"></strong></p>
    <button type="submit" :disabled="$form.submitting">Fechar pedido</button>
    <p data-form-status></p>
</form>
<script>
function pedido() {
    return {
        seq: 1,
        itens: [{ id: 1, produto: '', qtd: 1, preco: 0 }],
        total() { return this.itens.reduce((s, i) => s + i.qtd * i.preco, 0) },
    }
}
</script>

O total exibido é só conforto: o servidor recalcula com os preços do banco. O corpo enviado é {"itens":[{"produto":"…","quantidade":"2","preco":"19.90"}]} (preço canônico, sem "R$").

Formulário em etapas (wizard)

<form x-data="wizard()" x-form="{ hooks: { reveal: el => irPara(el) } }"
      action="/api/inscricao" method="post" novalidate>

    <section data-etapa="1" x-show="etapa === 1">
        <input name="nome" data-form-v="required|min:3">
        <input name="email" type="email" data-form-v="required|email">
        <button type="button" @click="avancar(['nome', 'email'])">Próximo</button>
    </section>

    <section data-etapa="2" x-show="etapa === 2">
        <input name="cpf" data-form-mask="cpf" data-form-v="required|cpf">
        <input name="nascimento" type="date" data-form-v="required|before:today">
        <button type="button" @click="etapa = 1">Voltar</button>
        <button type="button" @click="avancar(['cpf', 'nascimento'])">Próximo</button>
    </section>

    <section data-etapa="3" x-show="etapa === 3">
        <label><input type="checkbox" name="termos" value="1" data-form-v="accepted"> Aceito</label>
        <button type="button" @click="etapa = 2">Voltar</button>
        <button type="submit" :disabled="$form.submitting">Concluir</button>
    </section>
    <p data-form-status></p>
</form>

<script>
function wizard() {
    return {
        etapa: 1,
        async avancar(campos) {
            const resultados = await Promise.all(campos.map(nome => this.$form.validateField(nome)))
            if (resultados.every(r => r.valid)) this.etapa++
        },
        irPara(campo) {
            this.etapa = Number(campo.closest('[data-etapa]').dataset.etapa)
        },
    }
}
</script>

Cada "Próximo" valida só os campos da etapa. No envio final, todos são validados; se o erro (inclusive do servidor) estiver numa etapa escondida, o hook reveal volta para ela antes do foco.

Importação de planilha CSV

Requer alpine-form-inspectors (e alpine-form-chunks para arquivos muito grandes). O servidor responde 202 e processa em segundo plano.

<form x-data x-form action="/api/importacoes" method="post" enctype="multipart/form-data"
      data-form-job-interval="1s" data-form-job-persist novalidate>
    <input name="planilha" type="file" accept=".csv"
           data-form-v="required|ext:csv|file_max:50MB|csv|csv_headers:nome,email,valor"
           data-form-csv-encoding="utf-8">
    <div data-form-file-list="planilha"></div>

    <button type="submit" :disabled="$form.submitting">Importar</button>
    <button type="button" x-show="$form.processing" @click="$form.cancelJob()">Cancelar importação</button>

    <div data-form-progress>
        <progress data-form-progress-bar max="100" aria-label="Progresso da importação"></progress>
        <p data-form-status></p>
    </div>
    <p x-show="$form.success" x-text="$form.result && ('Importadas ' + $form.result.data.linhas + ' linhas.')"></p>
</form>

Formulário em modal

<div x-data="{ aberto: false }">
    <button type="button" @click="aberto = true">Novo contato</button>

    <dialog x-ref="dialogo" x-effect="aberto ? $refs.dialogo.showModal() : $refs.dialogo.close()" @close="aberto = false">
        <template x-if="aberto">
            <form x-data x-form action="/api/contatos" method="post"
                  @form:success="aberto = false; $dispatch('contato-criado', $event.detail.result.data)" novalidate>
                <input name="nome" data-form-v="required">
                <input name="email" type="email" data-form-v="required|email">
                <button type="submit" :disabled="$form.submitting">Salvar</button>
                <button type="button" @click="aberto = false">Fechar</button>
                <p data-form-status></p>
            </form>
        </template>
    </dialog>
</div>

Com x-if, o formulário é criado limpo a cada abertura e destruído (com todos os recursos liberados) ao fechar.
